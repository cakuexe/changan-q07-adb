@echo off
chcp 65001 >nul
title Отключение ADB при подключении устройства
echo ===============================================
echo Ожидание устройства...
adb wait-for-device
echo Устройство обнаружено! Выполняется команда отключения ADB...
echo ===============================================

adb shell setprop persist.usb.prim.adb_switch off

echo ===============================================
echo Команда выполнена!
echo Теперь ты можешь вводить свои команды вручную.
echo ===============================================
cmd /k
