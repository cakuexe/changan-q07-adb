import subprocess
import time
import threading
import tkinter as tk
from tkinter import ttk, scrolledtext
from datetime import datetime

MAX_WAIT = 300
CHECK_INTERVAL = 2

class ADBWatcherGUI:
    def __init__(self, root):
        self.root = root
        self.root.title("ADB Authorization Watcher")
        self.root.geometry("600x500")
        self.root.resizable(False, False)
        
        self.is_running = False
        self.start_time = None
        self.watcher_thread = None
        
        self.setup_ui()
        
    def setup_ui(self):
        # Header
        header_frame = tk.Frame(self.root, bg="#2c3e50", height=80)
        header_frame.pack(fill=tk.X)
        header_frame.pack_propagate(False)
        
        title_label = tk.Label(
            header_frame, 
            text="ADB Authorization Watcher",
            font=("Arial", 18, "bold"),
            bg="#2c3e50",
            fg="white"
        )
        title_label.pack(pady=25)
        
        # Status Frame
        status_frame = tk.Frame(self.root, padx=20, pady=15)
        status_frame.pack(fill=tk.X)
        
        tk.Label(status_frame, text="Status:", font=("Arial", 10, "bold")).pack(anchor=tk.W)
        
        self.status_label = tk.Label(
            status_frame,
            text="Hazır - Start düyməsinə basın",
            font=("Arial", 10),
            fg="#7f8c8d"
        )
        self.status_label.pack(anchor=tk.W, pady=5)
        
        # Progress bar
        self.progress = ttk.Progressbar(
            status_frame,
            mode='indeterminate',
            length=560
        )
        self.progress.pack(pady=10)
        
        # Log Frame
        log_frame = tk.Frame(self.root, padx=20)
        log_frame.pack(fill=tk.BOTH, expand=True)
        
        tk.Label(log_frame, text="Log:", font=("Arial", 10, "bold")).pack(anchor=tk.W)
        
        self.log_text = scrolledtext.ScrolledText(
            log_frame,
            height=12,
            font=("Consolas", 9),
            bg="#ecf0f1",
            fg="#2c3e50"
        )
        self.log_text.pack(fill=tk.BOTH, expand=True, pady=5)
        
        # Buttons Frame
        button_frame = tk.Frame(self.root, padx=20, pady=15)
        button_frame.pack(fill=tk.X)
        
        self.start_button = tk.Button(
            button_frame,
            text="Start Monitoring",
            command=self.start_monitoring,
            bg="#27ae60",
            fg="white",
            font=("Arial", 11, "bold"),
            padx=20,
            pady=8,
            relief=tk.FLAT,
            cursor="hand2"
        )
        self.start_button.pack(side=tk.LEFT, padx=5)
        
        self.stop_button = tk.Button(
            button_frame,
            text="Stop",
            command=self.stop_monitoring,
            bg="#e74c3c",
            fg="white",
            font=("Arial", 11, "bold"),
            padx=20,
            pady=8,
            relief=tk.FLAT,
            cursor="hand2",
            state=tk.DISABLED
        )
        self.stop_button.pack(side=tk.LEFT, padx=5)
        
        clear_button = tk.Button(
            button_frame,
            text="Clear Log",
            command=self.clear_log,
            bg="#95a5a6",
            fg="white",
            font=("Arial", 11, "bold"),
            padx=20,
            pady=8,
            relief=tk.FLAT,
            cursor="hand2"
        )
        clear_button.pack(side=tk.RIGHT, padx=5)
        
    def log_message(self, message, color="#2c3e50"):
        timestamp = datetime.now().strftime("%H:%M:%S")
        self.log_text.insert(tk.END, f"[{timestamp}] {message}\n")
        self.log_text.see(tk.END)
        
    def clear_log(self):
        self.log_text.delete(1.0, tk.END)
        
    def update_status(self, message, color="#7f8c8d"):
        self.status_label.config(text=message, fg=color)
        
    def get_adb_devices(self):
        try:
            output = subprocess.check_output(["adb", "devices"], stderr=subprocess.STDOUT)
            lines = output.decode().strip().split("\n")[1:]
            devices = []
            for line in lines:
                if line.strip():
                    parts = line.split()
                    device_id = parts[0]
                    status = parts[1] if len(parts) > 1 else "unknown"
                    devices.append((device_id, status))
            return devices
        except Exception as e:
            return []
    
    def monitor_devices(self):
        self.start_time = time.time()
        self.log_message("Monitoring başladı...")
        self.log_message("Cihaz gözlənilir...")
        
        while self.is_running:
            devices = self.get_adb_devices()
            
            if devices:
                for device_id, status in devices:
                    if status == "device":
                        self.root.after(0, self.update_status, 
                            "✓ Cihaz autorizasiya olundu!", "#27ae60")
                        self.log_message("=" * 50)
                        self.log_message("SUCCESS: Cihaz autorizasiya olundu!", "#27ae60")
                        self.log_message(f"Device ID: {device_id}")
                        self.log_message("=" * 50)
                        self.root.after(0, self.stop_monitoring)
                        return
                    
                    elif status == "unauthorized":
                        self.root.after(0, self.update_status,
                            f"⚠ Cihaz UNAUTHORIZED: {device_id}", "#e67e22")
                        self.log_message(f"Cihaz tapıldı, amma UNAUTHORIZED: {device_id}")
                        self.log_message(">> Telefonda RSA icazə pəncərəsi çıxmalıdır.")
                        self.log_message(">> Zəhmət olmasa 'Allow' edin.")
                    
                    else:
                        self.log_message(f"Cihaz görüldü, status: {status}")
            
            # Timeout check
            elapsed = time.time() - self.start_time
            remaining = MAX_WAIT - elapsed
            
            if remaining <= 0:
                self.root.after(0, self.update_status,
                    "✗ Maksimum gözləmə müddəti bitdi", "#e74c3c")
                self.log_message("=" * 50)
                self.log_message("FAILED: Maksimum gözləmə müddəti bitdi", "#e74c3c")
                self.log_message("=" * 50)
                self.log_message("Problemin həlli:")
                self.log_message("1) USB debugging aktiv olsun")
                self.log_message("2) Ekranı kiliddən çıxarın")
                self.log_message("3) RSA popup'da 'Allow' edin")
                self.log_message("4) Developer options → Revoke USB debugging authorizations")
                self.root.after(0, self.stop_monitoring)
                return
            
            self.root.after(0, self.update_status,
                f"Cihaz gözlənilir... ({int(remaining)}s qalıb)", "#3498db")
            
            time.sleep(CHECK_INTERVAL)
    
    def start_monitoring(self):
        self.is_running = True
        self.start_button.config(state=tk.DISABLED)
        self.stop_button.config(state=tk.NORMAL)
        self.progress.start(10)
        
        self.watcher_thread = threading.Thread(target=self.monitor_devices, daemon=True)
        self.watcher_thread.start()
        
    def stop_monitoring(self):
        self.is_running = False
        self.start_button.config(state=tk.NORMAL)
        self.stop_button.config(state=tk.DISABLED)
        self.progress.stop()
        
        if self.status_label.cget("text").startswith("Cihaz gözlənilir"):
            self.update_status("Dayandırıldı", "#7f8c8d")
            self.log_message("Monitoring dayandırıldı")

if __name__ == "__main__":
    root = tk.Tk()
    app = ADBWatcherGUI(root)
    root.mainloop()
